cmake_minimum_required(VERSION 3.20)
project(JavdexNativePlayback LANGUAGES CXX)

# Off by default: a separate experimental helper/FD bridge. The existing X11
# helper remains available and retains its runtime and dependency contract.
option(JAVDEX_BUILD_LINUX_DMABUF "Build experimental Linux shared-texture playback" OFF)
option(JAVDEX_BUILD_LINUX_CPU "Build experimental Linux CPU-frame playback" OFF)

# JS supplies native Windows paths; normalize before CMake interprets them as
# target source paths (a backslash followed by P is otherwise an invalid escape).
foreach(input ELECTRON_INCLUDE_DIR ELECTRON_NODE_LIBRARY ELECTRON_DELAY_LOAD_HOOK MPV_PREFIX PLAYBACK_OUTPUT_DIR)
  if(DEFINED ${input})
    file(TO_CMAKE_PATH "${${input}}" ${input})
  endif()
endforeach()

# Development adapter build. No dependency download and no runtime packaging;
# release inputs must still pass the separate reviewed runtime inventory gate.
if(NOT (WIN32 AND (MSVC OR MINGW)) AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This CMake entry builds the Windows MSVC/MinGW or Linux X11 adapter only")
endif()
foreach(input ELECTRON_INCLUDE_DIR MPV_PREFIX PLAYBACK_OUTPUT_DIR)
  if(NOT DEFINED ${input} OR "${${input}}" STREQUAL "")
    message(FATAL_ERROR "Missing ${input}")
  endif()
endforeach()
find_path(MPV_INCLUDE_DIR mpv/client.h PATHS "${MPV_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_library(MPV_LIBRARY NAMES mpv libmpv PATHS "${MPV_PREFIX}/lib" "${MPV_PREFIX}/lib/${CMAKE_LIBRARY_ARCHITECTURE}" NO_DEFAULT_PATH REQUIRED)
if(NOT EXISTS "${ELECTRON_INCLUDE_DIR}/node_api.h")
  message(FATAL_ERROR "Missing Electron headers")
endif()

if(WIN32)
  if(NOT EXISTS "${ELECTRON_NODE_LIBRARY}")
    message(FATAL_ERROR "Missing matching Electron node.lib")
  endif()
  if(MSVC AND NOT EXISTS "${ELECTRON_DELAY_LOAD_HOOK}")
    message(FATAL_ERROR "Missing node-gyp's Electron delay-load hook")
  endif()
  # node.lib imports node.exe, while Electron (including a renamed packaged
  # executable) exports those symbols itself. Compile the appropriate hook
  # into this module; a hook in another DLL cannot redirect its imports.
  if(MSVC)
    add_library(playback MODULE windowsPlayback.cpp "${ELECTRON_DELAY_LOAD_HOOK}")
    target_compile_options(playback PRIVATE /utf-8 /EHsc /W4)
    target_link_options(playback PRIVATE /DELAYLOAD:node.exe)
  else()
    # node-gyp's hook is guarded by _MSC_VER. The MinGW build supplies the
    # same host redirection through its own delayimp hook and LLVM's linker.
    add_library(playback MODULE windowsPlayback.cpp windowsElectronDelayHook.cpp)
    target_compile_options(playback PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(playback PRIVATE -Wl,--delayload=node.exe)
  endif()
  target_compile_definitions(playback PRIVATE _WIN32_WINNT=0x0A00 UNICODE _UNICODE HOST_BINARY="node.exe")
  target_link_libraries(playback PRIVATE "${ELECTRON_NODE_LIBRARY}" delayimp user32 gdi32 opengl32 comctl32 imm32 ole32 oleacc)
  add_executable(slider-feedback-test sliderFeedback.test.cpp)
  target_link_libraries(slider-feedback-test PRIVATE user32 comctl32)
  target_compile_features(slider-feedback-test PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(slider-feedback-test PRIVATE /UNDEBUG)
  else()
    target_compile_options(slider-feedback-test PRIVATE -UNDEBUG)
  endif()
  set_target_properties(slider-feedback-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PLAYBACK_OUTPUT_DIR}")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(slider-feedback-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${PLAYBACK_OUTPUT_DIR}")
  endforeach()
else()
  find_package(X11 REQUIRED)
  find_package(OpenGL REQUIRED COMPONENTS OpenGL GLX)
  add_executable(playback linuxX11Playback.cpp)
  target_compile_options(playback PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(playback PRIVATE X11::X11 X11::Xext OpenGL::OpenGL OpenGL::GLX)
  add_executable(x11-surface-test x11SurfaceClip.test.cpp)
  target_compile_features(x11-surface-test PRIVATE cxx_std_17)
  target_compile_options(x11-surface-test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  target_link_libraries(x11-surface-test PRIVATE X11::X11 X11::Xext)
  set_target_properties(x11-surface-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PLAYBACK_OUTPUT_DIR}")
  # Compiled with every Linux build; execute only under an explicitly supplied
  # isolated X display, not as a no-display/GL/hardware acceptance substitute.
  add_executable(x11-focus-test x11Focus.test.cpp)
  target_compile_features(x11-focus-test PRIVATE cxx_std_17)
  target_compile_options(x11-focus-test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  target_include_directories(x11-focus-test PRIVATE "${ELECTRON_INCLUDE_DIR}" "${MPV_INCLUDE_DIR}")
  target_link_libraries(x11-focus-test PRIVATE "${MPV_LIBRARY}" X11::X11 X11::Xext OpenGL::OpenGL OpenGL::GLX)
  set_target_properties(x11-focus-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PLAYBACK_OUTPUT_DIR}")
  add_executable(x11-controls-test x11Controls.test.cpp)
  target_compile_features(x11-controls-test PRIVATE cxx_std_17)
  target_compile_options(x11-controls-test PRIVATE -UNDEBUG)
  set_target_properties(x11-controls-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PLAYBACK_OUTPUT_DIR}")
endif()
target_compile_features(playback PRIVATE cxx_std_17)
target_include_directories(playback PRIVATE "${ELECTRON_INCLUDE_DIR}" "${MPV_INCLUDE_DIR}")
target_link_libraries(playback PRIVATE "${MPV_LIBRARY}")
set_target_properties(playback PROPERTIES PREFIX "" SUFFIX ".node"
  RUNTIME_OUTPUT_DIRECTORY "${PLAYBACK_OUTPUT_DIR}" LIBRARY_OUTPUT_DIRECTORY "${PLAYBACK_OUTPUT_DIR}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(playback PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${PLAYBACK_OUTPUT_DIR}"
    LIBRARY_OUTPUT_DIRECTORY_${config} "${PLAYBACK_OUTPUT_DIR}")
endforeach()

add_executable(mpv-core-test mpvCore.test.cpp)
target_compile_features(mpv-core-test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(mpv-core-test PRIVATE /utf-8 /EHsc /UNDEBUG)
else()
  target_compile_options(mpv-core-test PRIVATE -UNDEBUG)
endif()
target_include_directories(mpv-core-test PRIVATE "${ELECTRON_INCLUDE_DIR}" "${MPV_INCLUDE_DIR}")
target_link_libraries(mpv-core-test PRIVATE "${MPV_LIBRARY}")
set_target_properties(mpv-core-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PLAYBACK_OUTPUT_DIR}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(mpv-core-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${PLAYBACK_OUTPUT_DIR}")
endforeach()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set_target_properties(playback PROPERTIES OUTPUT_NAME "playback-helper" SUFFIX "" INSTALL_RPATH "$ORIGIN;$ORIGIN/lib")
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND JAVDEX_BUILD_LINUX_DMABUF)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DMABUF_EGL REQUIRED IMPORTED_TARGET egl)
  pkg_check_modules(DMABUF_GBM REQUIRED IMPORTED_TARGET gbm)
  pkg_check_modules(DMABUF_DRM REQUIRED IMPORTED_TARGET libdrm)
  add_executable(playback-dmabuf-helper linuxDmaBufPlayback.cpp)
  target_compile_features(playback-dmabuf-helper PRIVATE cxx_std_17)
  target_compile_options(playback-dmabuf-helper PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(playback-dmabuf-helper PRIVATE "${MPV_INCLUDE_DIR}")
  target_link_libraries(playback-dmabuf-helper PRIVATE "${MPV_LIBRARY}" OpenGL::OpenGL
    PkgConfig::DMABUF_EGL PkgConfig::DMABUF_GBM PkgConfig::DMABUF_DRM)
  set_target_properties(playback-dmabuf-helper PROPERTIES INSTALL_RPATH "$ORIGIN")

endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND (JAVDEX_BUILD_LINUX_DMABUF OR JAVDEX_BUILD_LINUX_CPU))
  # This thin addon owns descriptors and the socketpair only. Never link mpv or
  # graphics-driver libraries into Electron's main process.
  add_library(playback-fd MODULE linuxFdBridge.cpp)
  target_compile_features(playback-fd PRIVATE cxx_std_17)
  target_compile_options(playback-fd PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(playback-fd PRIVATE "${ELECTRON_INCLUDE_DIR}")
  set_target_properties(playback-fd PROPERTIES PREFIX "" SUFFIX ".node")
  add_executable(fd-transport-fixture linuxFdTransportFixture.cpp)
  target_compile_features(fd-transport-fixture PRIVATE cxx_std_17)
  target_compile_options(fd-transport-fixture PRIVATE -Wall -Wextra -Wpedantic)
  set(frame_targets playback-fd fd-transport-fixture)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND JAVDEX_BUILD_LINUX_DMABUF)
  add_executable(linux-dmabuf-test linuxDmaBuf.test.cpp)
  target_compile_features(linux-dmabuf-test PRIVATE cxx_std_17)
  target_compile_options(linux-dmabuf-test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  list(APPEND frame_targets playback-dmabuf-helper linux-dmabuf-test)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND JAVDEX_BUILD_LINUX_CPU)
  # A deterministic TTF avoids unsupported host default-font aliases. Only the
  # opt-in CPU helper receives these app-scoped assets; no host font install.
  get_filename_component(CPU_FONT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../resources/linux-cpu-fonts" ABSOLUTE)
  if(NOT EXISTS "${CPU_FONT_SOURCE}/NotoSans-Regular.ttf")
    message(FATAL_ERROR "Missing Linux CPU subtitle font")
  endif()
  add_custom_target(playback-cpu-fonts
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CPU_FONT_SOURCE}" "${PLAYBACK_OUTPUT_DIR}/fonts"
    VERBATIM)
  add_executable(playback-cpu-helper linuxCpuPlayback.cpp)
  add_dependencies(playback-cpu-helper playback-cpu-fonts)
  target_compile_features(playback-cpu-helper PRIVATE cxx_std_17)
  target_compile_options(playback-cpu-helper PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(playback-cpu-helper PRIVATE "${MPV_INCLUDE_DIR}")
  target_link_libraries(playback-cpu-helper PRIVATE "${MPV_LIBRARY}")
  set_target_properties(playback-cpu-helper PROPERTIES INSTALL_RPATH "$ORIGIN")
  add_executable(linux-cpu-frame-test linuxCpuFrame.test.cpp)
  target_compile_features(linux-cpu-frame-test PRIVATE cxx_std_17)
  target_compile_options(linux-cpu-frame-test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  list(APPEND frame_targets playback-cpu-helper linux-cpu-frame-test)
endif()

if(frame_targets)
  foreach(target ${frame_targets})
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PLAYBACK_OUTPUT_DIR}"
      LIBRARY_OUTPUT_DIRECTORY "${PLAYBACK_OUTPUT_DIR}")
    foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
      set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${PLAYBACK_OUTPUT_DIR}"
        LIBRARY_OUTPUT_DIRECTORY_${config} "${PLAYBACK_OUTPUT_DIR}")
    endforeach()
  endforeach()
endif()
