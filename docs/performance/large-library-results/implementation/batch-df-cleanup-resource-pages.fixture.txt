import { createScanCleanupPages } from '/Users/nut/Project/JavdexLabs/Javdex/src/main/scanner/scanCleanupPages.ts'
it('walks 100000 real resource rows without deletion using bounded monotonic pages',async()=>{
  const root=db.prepare('SELECT id FROM media_library_roots WHERE library_id=?').get(libraryId) as {id:number}
  db.prepare(`WITH RECURSIVE n(id) AS (SELECT 1 UNION ALL SELECT id+1 FROM n WHERE id<100000)
    INSERT INTO video_resources(library_id,video_id,root_id,kind,locator,resource_key,source_identity)
    SELECT ?,1,?,'local','/resource/'||id,'key-'||id,'source-'||id FROM n`).run(libraryId,root.id)
  const legacyPlan=db.prepare('EXPLAIN QUERY PLAN SELECT id FROM video_resources WHERE library_id=? AND id>? AND id<=? ORDER BY id LIMIT 128').all(libraryId,0,100000)
  const fixedPlan=db.prepare('EXPLAIN QUERY PLAN SELECT id FROM video_resources NOT INDEXED WHERE library_id=? AND id>? AND id<=? ORDER BY id LIMIT 128').all(libraryId,0,100000)
  const pages=createScanCleanupPages(db,libraryId)
  let rows=0,queries=0,beats=0,last=performance.now(),maxGap=0
  const interval=setInterval(()=>{const now=performance.now();maxGap=Math.max(maxGap,now-last);last=now;beats++},1)
  const start=performance.now()
  try {
    await pages.each('resources',new AbortController().signal,fn=>db.transaction(fn)(),ids=>{
      assert.ok(ids.length<=128);rows+=ids.length;queries++
    },()=>assert.equal(db.inTransaction,false))
    const elapsedMs=performance.now()-start
    maxGap=Math.max(maxGap,performance.now()-last)
    assert.equal(rows,100000);assert.equal(queries,783);assert.ok(beats>0)
    assert.equal((db.prepare('SELECT COUNT(*) AS n FROM video_resources').get() as {n:number}).n,100000)
    console.log(JSON.stringify({rows,queries,beats,elapsedMs,maxGap,legacyPlan,fixedPlan,notes:'Synthetic file-backed DB. No deletion, audit, filesystem checks or finalization. Only the fixed plan is timed.'}))
  }finally{clearInterval(interval);pages.dispose()}
})
